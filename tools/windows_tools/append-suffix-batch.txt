Remplacer `MON_SUFFIXE` par le texte de votre choix (gardez les guillemets et `$_.Extension`)

---

Get-ChildItem | Where-Object { -not $_.PSIsContainer } | ForEach-Object {
    $newname = $_.BaseName + "MON_SUFFIXE" + $_.Extension
    Rename-Item $_.FullName -NewName $newname
}


---

Get-ChildItem | Where-Object { -not $_.PSIsContainer } | ForEach-Object {
    $newname = $_.BaseName + "_colorized" + $_.Extension
    Rename-Item $_.FullName -NewName $newname
}
